Where the matches live

The answer is a small model class. HomeScreen creates one in setup and passes it to the cards as a prop. follows, draft and open stay plain setup variables, because they describe the UI. The matches are fetched data with rules for staleness and dedupe, and more than one component reads them. Remix's interactivity guide draws the line in the same place, in "Keep application logic outside components".

The model

This is about 50 lines. TypedEventTarget comes from remix/component, and the href comes from the shared route contract in app/routes.ts. fetch and now are injectable so the tests can control them.

app/actions/public/clubs/matches.ts new

import { TypedEventTarget } from 'remix/component';

import { routes } from '../../../routes.ts';
import type { ClubCard } from '../../../lib/club-card.ts';

export const MATCHES_MAX_AGE_MS = 60 * 60 * 1000;

export type MatchesEntry =
  | { status: 'loading'; card?: ClubCard }   // card = previous data during a refresh
  | { status: 'ready'; card: ClubCard }
  | { status: 'error' };

export class MatchesChange extends Event {
  constructor(readonly clubId: number) {
    super('change');
  }
}

type Options = { fetch?: typeof fetch; now?: () => number };

export class MatchesModel extends TypedEventTarget<{ change: MatchesChange }> {
  #entries = new Map<number, MatchesEntry>();
  #fetch: typeof fetch;
  #now: () => number;

  constructor(seed: ClubCard[] = [], options: Options = {}) {
    super();
    this.#fetch = options.fetch ?? globalThis.fetch.bind(globalThis);
    this.#now = options.now ?? Date.now;
    for (const card of seed) this.#entries.set(card.teamId, { status: 'ready', card });
  }

  get(clubId: number): MatchesEntry | undefined {
    return this.#entries.get(clubId);
  }

  load(clubId: number): void {
    const entry = this.#entries.get(clubId);
    if (entry?.status === 'loading') return;
    if (entry?.status === 'ready' && this.#now() - entry.card.fetchedAt < MATCHES_MAX_AGE_MS) return;

    const previous = entry?.status === 'ready' ? entry.card : undefined;
    this.#set(clubId, { status: 'loading', card: previous });

    this.#request(clubId).then(
      (card) => this.#set(clubId, { status: 'ready', card }),
      () => this.#set(clubId, previous ? { status: 'ready', card: previous } : { status: 'error' }),
    );
  }

  async #request(clubId: number): Promise<ClubCard> {
    const res = await this.#fetch(routes.clubMatches.href({ clubId }));
    if (!res.ok) throw new Error(`matches ${clubId}: ${res.status}`);
    return res.json();
  }

  #set(clubId: number, entry: MatchesEntry) {
    this.#entries.set(clubId, entry);
    this.dispatchEvent(new MatchesChange(clubId));
  }
}

clubs/club-card.tsx new · data part only

type Props = { club: Club; matches: MatchesModel };

export function ClubCard(handle: Handle<Props>) {
  const { matches } = handle.props;   // same instance every render

  matches.addEventListener(
    'change',
    (event) => {
      if (event.clubId === handle.props.club.id) handle.update();
    },
    { signal: handle.signal },          // unsubscribes on unmount
  );

  return () => {
    const { club } = handle.props;
    const entry = matches.get(club.id);
    const card = entry?.status === 'error' ? undefined : entry?.card;

    return (
      <article>
        <ClubCrest crest={club.crest} /> {club.name}
        {card ? <Form card={card} /> : <Skeleton />}
        {entry?.status === 'error' && (
          <button mix={on('click', () => matches.load(club.id))}>Retry</button>
        )}
      </article>
    );
  };
}

Every card hears every change and ignores the ones for other ids. With a handful of followed clubs that check costs nothing. Per-id event names like change:57 would lose the typed event map.

The model across one visit
